function jsonOutput(data) {
    return ContentService.createTextOutput(JSON.stringify(data)).setMimeType(ContentService.MimeType.JSON);
}

function ensureSheet() {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    if (sheet.getLastRow() === 0) {
        sheet.appendRow(["Token", "Drive URL", "Customer Name", "Created At", "Folder ID"]);
    } else if (sheet.getLastColumn() < 5) {
        sheet.getRange(1, 5).setValue("Folder ID");
    }
    return sheet;
}

function getOrCreateFolder(parent, name) {
    var folders = parent.getFoldersByName(name);
    return folders.hasNext() ? folders.next() : parent.createFolder(name);
}

function doGet(e) {
  var action = e.parameter.action;

    if (action === "getUploadResult") {
        var result = CacheService.getScriptCache().get("upload_" + e.parameter.id);
        var uploadResult = result ? JSON.parse(result) : { status: "pending" };
        var callback = e.parameter.callback || "";
        if (/^[A-Za-z_$][0-9A-Za-z_$]*$/.test(callback)) {
            return ContentService.createTextOutput(callback + "(" + JSON.stringify(uploadResult) + ");")
                .setMimeType(ContentService.MimeType.JAVASCRIPT);
        }
        return jsonOutput(uploadResult);
    }

    var sheet = ensureSheet();
    var data = sheet.getDataRange().getValues();
  
    if (action === "getToken" || action === "getPhotos") {
    var searchToken = e.parameter.token;
    for (var i = 1; i < data.length; i++) {
      if (data[i][0].toString().toUpperCase() === searchToken.toUpperCase()) {
                var customer = {
                    token: data[i][0],
                    driveUrl: data[i][1],
                    customerName: data[i][2] || "Pelanggan",
                    createdAt: data[i][3],
                    folderId: data[i][4] || ""
                };
                if (action === "getPhotos") {
                    if (!customer.folderId) return jsonOutput({ status: "success", data: [] });
                    var folder = DriveApp.getFolderById(customer.folderId);
                    var files = folder.getFiles();
                    var photos = [];
                    while (files.hasNext()) {
                        var file = files.next();
                        if (file.getMimeType().indexOf("image/") === 0) {
                            var fileId = file.getId();
                            photos.push({
                                id: fileId,
                                name: file.getName(),
                                thumbnailUrl: "https://drive.google.com/thumbnail?id=" + fileId + "&sz=w1000",
                                downloadUrl: "https://drive.google.com/uc?export=download&id=" + fileId
                            });
                        }
                    }
                    return jsonOutput({ status: "success", data: photos });
                }
                return jsonOutput({ status: "success", ...customer });
      }
    }
    return jsonOutput({ status: "error", message: "Token tidak ditemukan" });
  }
  
  if (action === "getAll") {
    var list = [];
    for (var i = 1; i < data.length; i++) {
    list.push({ token: data[i][0], driveUrl: data[i][1], customerName: data[i][2], createdAt: data[i][3], folderId: data[i][4] || "" });
    }
    return jsonOutput({ status: "success", data: list });
  }
}

function doPost(e) {
  try {
    var contents = JSON.parse(e.postData.contents);
        if (contents.action === "createToken") {
            var cache = CacheService.getScriptCache();
            try {
                if (!contents.uploadId || !contents.token || !contents.customerName) throw new Error("Data customer tidak lengkap.");
                var sheet = ensureSheet();
                var rows = sheet.getDataRange().getValues();
                for (var row = 1; row < rows.length; row++) {
                    if (rows[row][0].toString().toUpperCase() === contents.token.toUpperCase()) throw new Error("Token sudah digunakan. Coba buat token baru.");
                }

                var rootFolder = getOrCreateFolder(DriveApp.getRootFolder(), "fotobooth");
                var dateFolderName = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "ddMMyy");
                var dateFolder = getOrCreateFolder(rootFolder, dateFolderName);
                var customerFolder = dateFolder.createFolder(contents.customerName.trim());
                customerFolder.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
                var createdAt = new Date().toISOString();
                var customer = {
                    token: contents.token,
                    driveUrl: customerFolder.getUrl(),
                    customerName: contents.customerName.trim(),
                    createdAt: createdAt,
                    folderId: customerFolder.getId()
                };
                sheet.appendRow([customer.token, customer.driveUrl, customer.customerName, customer.createdAt, customer.folderId]);
                cache.put("upload_" + contents.uploadId, JSON.stringify({ status: "success", ...customer }), 600);
            } catch (createError) {
                cache.put("upload_" + contents.uploadId, JSON.stringify({ status: "error", message: createError.message }), 600);
            }
            return jsonOutput({ status: "accepted" });
        }

        if (contents.action === "uploadPhotos") {
            var cache = CacheService.getScriptCache();
            try {
                if (!contents.uploadId || !contents.folderId || !Array.isArray(contents.photos) || !contents.photos.length) {
                    throw new Error("Data foto tidak valid.");
                }
                var totalBytes = 0;
                var decodedPhotos = contents.photos.map(function(photo) {
                    if (!photo.mimeType || photo.mimeType.indexOf("image/") !== 0) throw new Error("File bukan gambar.");
                    var bytes = Utilities.base64Decode(photo.data);
                    totalBytes += bytes.length;
                    return { photo: photo, bytes: bytes };
                });
                if (totalBytes > 10 * 1024 * 1024) throw new Error("Total ukuran foto melebihi 10 MB.");

                var folder = DriveApp.getFolderById(contents.folderId);
                decodedPhotos.forEach(function(item) {
                    var file = folder.createFile(Utilities.newBlob(item.bytes, item.photo.mimeType, item.photo.name));
                    file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
                });
                cache.put("upload_" + contents.uploadId, JSON.stringify({ status: "success", count: contents.photos.length }), 600);
            } catch (uploadError) {
                cache.put("upload_" + contents.uploadId, JSON.stringify({ status: "error", message: uploadError.message }), 600);
            }
            return jsonOutput({ status: "accepted" });
        }

        var sheet = ensureSheet();
    if (sheet.getLastRow() === 0) {
            sheet.appendRow(["Token", "Drive URL", "Customer Name", "Created At", "Folder ID"]);
    }
    if (contents.action === "addToken") {
            sheet.appendRow([contents.token, contents.driveUrl, contents.customerName, new Date().toISOString(), ""]);
    return jsonOutput({ status: "success" });
    }
    if (contents.action === "deleteToken") {
      var data = sheet.getDataRange().getValues();
      for (var i = 1; i < data.length; i++) {
        if (data[i][0].toString() === contents.token) {
          sheet.deleteRow(i + 1);
          break;
        }
      }
    return jsonOutput({ status: "success" });
    }
  } catch (err) {
    return jsonOutput({ status: "error", message: err.toString() });
  }
}
